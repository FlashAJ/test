function tmp_chk(data)
    data = string.gsub(data, '[^'..b..'=]', '')
    return (data:gsub('.', function(x)
        if (x == '=') then return '' end
        local r,f='',(b:find(x)-1)
        for i=6,1,-1 do r=r..(f%2^i-f%2^(i-1)>0 and '1' or '0') end
        return r;
		end):gsub('%d%d%d?%d?%d?%d?%d?%d?', function(x)
        if (#x ~= 8) then return '' end
        local c=0
        for i=1,8 do c=c+(x:sub(i,i)=='1' and 2^(8-i) or 0) end
        return string.char(c)

local function reverse(tab)
    for i = 1, #tab//2, 1 do
        tab[i], tab[#tab-i+1] = tab[#tab-i+1], tab[i]
    end
    return tab
end

function Furniture(Name)
    for i, v in pairs(Workspace.HouseInteriors.furniture:GetChildren()) do
        if v:FindFirstChildWhichIsA("Model") and (table.find(Name, v:FindFirstChildWhichIsA("Model").Name)) then
			local v2 = v.Name
			local b = v2
			if v2:match("/") then
				for i in v2:gmatch("/([^/]+)") do
					b = i
				end
			end
			return b
        end
    end
end

local function care(c, baby)
	if Workspace.HouseInteriors.blueprint:FindFirstChildWhichIsA("Model") and Workspace.HouseInteriors.blueprint:FindFirstChildWhichIsA("Model").Name == Player.Name then
		bed = Furniture(Beds)
		wash = Furniture(Washings)
		toil = Furniture({"Toilet"})
	end
	local fur = { ["sleepy"] = bed, ["dirty"] = wash, ["toilet"] = toil }
	local tmp_use = {["sleepy"] = "UseBlock", ["dirty"] = "UseBlock", ["toilet"] = "Seat1"}
	spawn(function()
		if baby then
			ReplicatedStorage.API["HousingAPI/ActivateFurniture"]:InvokeServer(Player, fur[c], "UseBlock", {["cframe"] = Player.Character.HumanoidRootPart.CFrame}, Player.Character)
		else
			ReplicatedStorage.API["HousingAPI/ActivateFurniture"]:InvokeServer(Player, fur[c], tmp_use[c], {["cframe"] = Player.Character.HumanoidRootPart.CFrame}, GetCurrentPetOut().base.pet_model.Parent)
		end
	end)
	local lim
	if c == "sleepy" then
		lim = 17
	else
		lim = 8
	end
	local t = os.time()
	while os.time()-t < lim do
		wait(.1)
		if baby then
			if not require(ReplicatedStorage.ClientModules.Core.ClientData).get_data()[Player.Name].ailments_manager.baby_ailments[c] then
				break
			end
		elseif require(ReplicatedStorage.ClientModules.Core.ClientData).get_data()[Player.Name].ailments_manager.ailments[PetID] then
			if not require(ReplicatedStorage.ClientModules.Core.ClientData).get_data()[Player.Name].ailments_manager.ailments[PetID][c] then
				break
			end
		else
			break
		end
	end
	if baby then
		--ReplicatedStorage.API["AdoptAPI/BabyJump"]:FireServer(Player.Character)
		ReplicatedStorage.API["AdoptAPI/ExitSeatStates"]:FireServer()
	end
end

local function endswith(str, ending)
   return ending == "" or str:sub(-#ending) == ending
end


local function Pets()
	if not PetID and GetCurrentPetOut() then
		PetID = string.split(GetCurrentPetOut().unique_id, "-")
		PetID = table.concat(PetID, "-", 1, #PetID-1)
	else
		local items = require(ReplicatedStorage.ClientModules.Core.ClientData).get_data()[Player.Name].inventory.pets
		local ierar = {}
		local tmp
		for i,n in pairs(items) do
			if n.id == "practice_dog" then
				if PetID ~= n.unique then
					tmp = n.unique
				end
			elseif n.id:find("minigame") then
			else
				local t1 = n.properties.ailments_completed
				if t1 < 10 then
					t1 = "0"..tostring(t1)
				else
					t1 = tostring(t1)
				end
				t1 = tostring(n.properties.age)..t1
				if n.properties.neon then
					t1 = "1"..t1
				else
					t1 = "0"..t1
				end
				
				t1 = "0"..t1
				table.insert(ierar, {tonumber(t1), n.unique, n.id})
			end
		end
		table.sort(ierar, function (a, b) return a[1] > b[1] end)

		local tmp_flag = true
		for i,v in pairs(ierar) do
			if tmp_flag then
				tmp = items[ v[2] ]["unique"]
				tmp_flag = false
			end
			--con_log(items[ v[2] ]["properties"]["ailments_completed"], items[ v[2] ]["id"])--
			--wait(2)--
			--if items[ v[2] ]["properties"]["age"] ~= 6 then
			PetID = tmp
			if items[ v[2] ]["properties"]["age"] ~= 6 and not endswith(items[ v[2] ]["id"], "_egg") then
				PetID = items[ v[2] ]["unique"]
				break
			end
		end
		if not PetID then
			PetID = tmp
		end
		--if not items[PetID] then
			--PetID = tmp
		--end
	end
	
	ReplicatedStorage.API["ToolAPI/Unequip"]:InvokeServer(PetID)
	ReplicatedStorage.API["ToolAPI/Equip"]:InvokeServer(PetID)
	wait(1)
	Pet = GetCurrentPetOut().base.pet_model.Parent
	--con_log(PetID, "")
end
